# The assertion that failed in CI, reproduced

autosave slot must hold this seed's world from an even day, at most three days before the day the world
stopped on and not after it. (tests/e2e/resilience.spec.ts as it stood at commits 55dc9db and de2f0e1:
`s.day % 2 === 0 && s.day >= Math.max(2, today - 3) && s.day <= today`.)

5 of 10 fail.

Every failure is a save from the very day the world stopped on, when that day is odd: the newest save
there could be, refused for the day it was made. In eight of the ten runs the save was made at the tick the
world stopped at: the page asked for it on seeing a second morning, and by the time the request reached the
worker the world had run to its stop. (Run 8 also stopped on an odd day; there the save was made during day
10, before the world ran on into day 11, and the assertion held.)

The CI run that failed (45690303999, desktop, commit 55dc9db) stopped on day 9. Its log does not say what
the slot held. What D-050 asserts instead holds in every one of these ten runs, on their own numbers: each
slot is from no earlier than the second morning the page had seen while time ran (`sawTarget.tick` below), and
none is from after the world stopped (`stop`).

## The probe (node lag.cjs http://127.0.0.1:3298 10 200)

// Scratch probe (not part of the suite): what the autosave slot holds after playing at the fastest speed
// and pausing, when the page's main thread is slowed. Usage: node lag.cjs <baseURL> <rate> [target]
const { chromium } = require("playwright");
const base = process.argv[2];
const rate = Number(process.argv[3] ?? 1);
const target = Number(process.argv[4] ?? 200);

function autosave(page) {
  return page.evaluate(
    () =>
      new Promise((resolve) => {
        const open = indexedDB.open("tabula-rasa");
        open.onerror = () => resolve(null);
        open.onsuccess = () => {
          const db = open.result;
          if (!db.objectStoreNames.contains("saves")) { db.close(); resolve(null); return; }
          const get = db.transaction("saves", "readonly").objectStore("saves").get("autosave");
          get.onerror = () => resolve(null);
          get.onsuccess = () => { const r = get.result; db.close(); resolve(r ? { seed: r.seed, day: r.day, tick: r.tick } : null); };
        };
      }),
  );
}
const tick = async (page) => Number(((await page.getByTestId("tick").textContent()) ?? "").replace(/\D+/g, ""));

(async () => {
  const browser = await chromium.launch();
  const context = await browser.newContext({ viewport: { width: 1366, height: 768 } });
  const page = await context.newPage();
  const errors = [];
  page.on("pageerror", (e) => errors.push(String(e)));
  await page.goto(`${base}/?q=low&pr=0.4`);
  await page.getByTestId("begin").click();
  await page.getByTestId("scenario-first_light").click();
  await page.getByTestId("seed-input").fill("7");
  await page.getByTestId("create-world").click();
  await page.getByTestId("world-label").waitFor({ timeout: 60000 });
  await page.getByTestId("coach-skip").click();
  const cdp = await context.newCDPSession(page);
  await cdp.send("Emulation.setCPUThrottlingRate", { rate });
  const t0 = Date.now();
  await page.getByTestId("speed-5").click();
  if ((await page.getByTestId("play-toggle").getAttribute("data-playing")) !== "true") await page.getByTestId("play-toggle").click();
  const tPlay = Date.now();
  let seen = 0;
  for (;;) {
    seen = await tick(page);
    if (seen > target) break;
    await new Promise((r) => setTimeout(r, 250));
    if (Date.now() - tPlay > 240000) throw new Error("never got there");
  }
  const tSeen = Date.now();
  await page.getByTestId("play-toggle").click();
  const tClicked = Date.now();
  await page.waitForFunction(() => document.querySelector('[data-testid="play-toggle"]')?.getAttribute("data-playing") === "false", null, { timeout: 120000 });
  const tPaused = Date.now();
  const stop = await tick(page);
  const today = Math.floor(stop / 96);
  const log = [];
  for (let i = 0; i < 12; i++) {
    const s = await autosave(page);
    log.push(s ? `${s.day}@${s.tick}` : "none");
    await new Promise((r) => setTimeout(r, 1000));
  }
  const final = await autosave(page);
  const ok = final !== null && final.seed === 7 && final.day % 2 === 0 && final.day >= Math.max(2, today - 3) && final.day <= today;
  console.log(JSON.stringify({ rate, playMs: tPlay - t0, sawTarget: { tick: seen, afterMs: tSeen - tPlay }, clickMs: tClicked - tSeen, untilPausedShownMs: tPaused - tClicked, stop, today, slot: log, oldPredicate: ok, errors }));
  await browser.close();
})().catch((e) => { console.error(e); process.exit(1); });
