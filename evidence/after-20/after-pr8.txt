1. A tapped card eases open
---------------------------

On a phone a citizen's card opens drawn back and is opened in full by a tap on its head. Before pull request 8,
letting go of the head after any touch set a transition for the card's position on the element itself, and
for a fifth of a second that put aside the rule that eases the card's height.

What was asked of the browser: its own transition events on the card (transitionrun, -start, -end) from just
before each touch, and the card's height at every frame it then drew.

The deployed site, after the merge (node ease.cjs https://tabula-rasa-bay.vercel.app; the script is in 4):

  a tap on its head: 287px -> 490px; the browser's own events: transitionrun:max-height, transitionstart:max-height, transitionend:max-height; heights at the frames drawn: 287 287 490 490 490 (0 in between)
  its arrow: 490px -> 287px; the browser's own events: transitionrun:max-height, transitionstart:max-height, transitionend:max-height; heights at the frames drawn: 490 490 287 287 287 (0 in between)
  a tap on its head: 287px -> 490px; the browser's own events: transitionrun:max-height, transitionstart:max-height, transitionend:max-height; heights at the frames drawn: 287 287 490 490 490 (0 in between)

A build of this tree with the one line of src/ui/game/swipe.ts put back as it was before pull request 8:

  a tap on its head: 287px -> 490px; the browser's own events: none; heights at the frames drawn: 490 490 490 490 490 (0 in between)
  its arrow: 490px -> 287px; the browser's own events: transitionrun:max-height, transitionstart:max-height, transitionend:max-height; heights at the frames drawn: 490 490 287 287 (0 in between)
  a tap on its head: 287px -> 490px; the browser's own events: none; heights at the frames drawn: 490 490 490 490 490 (0 in between)

So: with the old line a tap on the head ran no transition and the card was at its full height at the first
reading, while the arrow — a button, which the old line never touched — eased. On the deployed site both
ease. The heights "at the frames drawn" show nothing in between in either case: a quarter-second transition
falls between two frames here. That the easing is SEEN is therefore not shown by this file; that the browser
runs it, and did not, is.

ease.cjs

// Scratch: does a tapped card ease open, or jump? Its height, frame by frame, after a tap on its head. node ease.cjs <base>
const { chromium } = require("playwright");
const base = process.argv[2];
(async () => {
  const browser = await chromium.launch();
  const context = await browser.newContext({ viewport: { width: 390, height: 844 }, isMobile: true, hasTouch: true, deviceScaleFactor: 2 });
  const page = await context.newPage();
  const tick = async () => Number(((await page.getByTestId("tick").textContent()) ?? "").replace(/\D+/g, ""));
  await page.goto(`${base}/?q=low&pr=0.5`);
  await page.getByTestId("begin").click();
  await page.getByTestId("scenario-first_light").click();
  await page.getByTestId("seed-input").fill("7");
  await page.getByTestId("create-world").click();
  await page.getByTestId("world-label").waitFor({ state: "attached", timeout: 60000 });
  await page.getByTestId("coach-skip").click();
  await page.getByTestId("speed-3").click();
  await page.getByTestId("play-toggle").click();
  for (;;) { if ((await tick()) > 130) break; await page.waitForTimeout(300); }
  await page.getByTestId("play-toggle").click();
  await page.waitForFunction(() => document.querySelector('[data-testid="play-toggle"]')?.getAttribute("data-playing") === "false");
  const target = await page.evaluate(() => { const w = window.innerWidth, h = window.innerHeight; for (let id = 0; id < 50; id++) { const p = window.__tr?.screenOf(id); if (p && p.x > w * 0.12 && p.x < w * 0.88 && p.y > h * 0.26 && p.y < h * 0.72) return p; } return null; });
  await page.touchscreen.tap(target.x, target.y);
  await page.getByTestId("inspector-name").waitFor({ timeout: 8000 });
  await page.waitForTimeout(1500);
  // what the browser says of its own transitions on the card, and the card's height at every frame it draws
  await page.getByTestId("inspector").evaluate((el) => {
    window.__log = [];
    for (const type of ["transitionrun", "transitionstart", "transitionend", "transitioncancel"]) el.addEventListener(type, (e) => { if (e.target === el) window.__log.push(`${type}:${e.propertyName}`); });
    new MutationObserver(() => {
      const t0 = performance.now();
      window.__heights = [];
      const read = () => { window.__heights.push(Math.round(el.getBoundingClientRect().height)); if (performance.now() - t0 < 700) requestAnimationFrame(read); };
      read();
    }).observe(el, { attributes: true, attributeFilter: ["data-compact"] });
  });
  for (const how of ["a tap on its head", "its arrow", "a tap on its head"]) {
    await page.evaluate(() => { window.__log = []; window.__heights = []; });
    const before = await page.getByTestId("inspector").evaluate((el) => Math.round(el.getBoundingClientRect().height));
    if (how === "its arrow") await page.getByTestId("inspector-toggle").tap(); else await page.getByTestId("inspector-name").tap();
    await page.waitForTimeout(1500);
    const r = await page.evaluate(() => ({ log: window.__log, heights: window.__heights }));
    const after = await page.getByTestId("inspector").evaluate((el) => Math.round(el.getBoundingClientRect().height));
    const between = [...new Set(r.heights)].filter((h) => h !== before && h !== after);
    console.log(`${how}: ${before}px -> ${after}px; the browser's own events: ${r.log.join(", ") || "none"}; heights at the frames drawn: ${r.heights.join(" ")} (${between.length} in between)`);
  }
  await browser.close();
})().catch((e) => { console.error(String(e).slice(0, 900)); process.exit(1); });

// Scratch: what pull request 8 changed, looked for on a running site. node after8.cjs <base>
const { chromium } = require("playwright");
const base = process.argv[2];
(async () => {
  const browser = await chromium.launch();
  // 1. phone: a tap on the card's head leaves the card its own transition (it eases open)
  {
    const context = await browser.newContext({ viewport: { width: 390, height: 844 }, isMobile: true, hasTouch: true, deviceScaleFactor: 2 });
    const page = await context.newPage();
    const tick = async () => Number(((await page.getByTestId("tick").textContent()) ?? "").replace(/\D+/g, ""));
    await page.goto(`${base}/?q=low&pr=0.5`);
    await page.getByTestId("begin").click();
    await page.getByTestId("scenario-first_light").click();
    await page.getByTestId("seed-input").fill("7");
    await page.getByTestId("create-world").click();
    await page.getByTestId("world-label").waitFor({ state: "attached", timeout: 60000 });
    await page.getByTestId("coach-skip").click();
    await page.getByTestId("speed-3").click();
    await page.getByTestId("play-toggle").click();
    for (;;) { if ((await tick()) > 130) break; await page.waitForTimeout(300); }
    await page.getByTestId("play-toggle").click();
    await page.waitForFunction(() => document.querySelector('[data-testid="play-toggle"]')?.getAttribute("data-playing") === "false");
    const target = await page.evaluate(() => { const w = window.innerWidth, h = window.innerHeight; for (let id = 0; id < 50; id++) { const p = window.__tr?.screenOf(id); if (p && p.x > w * 0.12 && p.x < w * 0.88 && p.y > h * 0.26 && p.y < h * 0.72) return p; } return null; });
    await page.touchscreen.tap(target.x, target.y);
    await page.getByTestId("inspector-name").waitFor({ timeout: 8000 });
    await page.waitForTimeout(1500);
    const compactBefore = await page.getByTestId("inspector").getAttribute("data-compact");
    await page.getByTestId("inspector-name").tap();
    const inline = await page.getByTestId("inspector").evaluate((el) => el.style.transition);
    await page.waitForTimeout(600);
    const compactAfter = await page.getByTestId("inspector").getAttribute("data-compact");
    console.log(`phone: card drawn back before the tap: ${compactBefore === "true"}; open after it: ${compactAfter !== "true"}; inline transition left on the card by the tap: ${JSON.stringify(inline)} (empty means it eases open by its own rule)`);
    await context.close();
  }
  // 2. desktop: follow someone, then go to where a Chronicle entry about them happened: the button and the camera agree
  {
    const context = await browser.newContext({ viewport: { width: 1366, height: 768 } });
    const page = await context.newPage();
    const tick = async () => Number(((await page.getByTestId("tick").textContent()) ?? "").replace(/\D+/g, ""));
    await page.goto(`${base}/?q=low&pr=0.5`);
    await page.getByTestId("begin").click();
    await page.getByTestId("scenario-first_light").click();
    await page.getByTestId("seed-input").fill("7");
    await page.getByTestId("create-world").click();
    await page.getByTestId("world-label").waitFor({ state: "attached", timeout: 60000 });
    await page.getByTestId("coach-skip").click();
    await page.getByTestId("speed-4").click();
    await page.getByTestId("play-toggle").click();
    for (;;) { if ((await tick()) > 300) break; await page.waitForTimeout(300); }
    await page.getByTestId("play-toggle").click();
    await page.waitForFunction(() => document.querySelector('[data-testid="play-toggle"]')?.getAttribute("data-playing") === "false");
    // through the Chronicle: an entry selects who did it
    await page.keyboard.press("c");
    await page.getByTestId("chronicle").waitFor({ timeout: 8000 });
    const entries = page.getByTestId("chronicle-entries").locator('button[title="Go to where this happened"]');
    const n = await entries.count();
    let found = false;
    for (let i = 0; i < Math.min(n, 40) && !found; i++) {
      await entries.nth(i).click();
      found = await page.getByTestId("inspector-name").waitFor({ state: "visible", timeout: 1500 }).then(() => true).catch(() => false);
    }
    if (!found) throw new Error("no Chronicle entry selected a citizen");
    const who = await page.evaluate(() => window.__tr.selected()?.id ?? null);
    await page.getByTestId("follow").click();
    await page.waitForTimeout(800);
    const before = { button: await page.getByTestId("follow").getAttribute("aria-pressed"), camera: await page.evaluate(() => window.__tr.following()) };
    // an entry about the same citizen, somewhere: "Only <name>" narrows the list to them
    const only = page.getByRole("button", { name: /^Only / });
    if (await only.count()) await only.click();
    await page.waitForTimeout(400);
    const mine = page.getByTestId("chronicle-entries").locator('button[title="Go to where this happened"]');
    let same = false;
    for (let i = 0; i < Math.min(await mine.count(), 30) && !same; i++) {
      await mine.nth(i).click();
      await page.waitForTimeout(500);
      same = (await page.evaluate(() => window.__tr.selected()?.id ?? null)) === who;
      if (same) break;
    }
    const after = { button: await page.getByTestId("follow").getAttribute("aria-pressed"), camera: await page.evaluate(() => window.__tr.following()) };
    console.log(`desktop: following citizen ${who}: button ${before.button}, camera ${before.camera}; after going to an entry about ${same ? "the same citizen" : "someone (no entry about the same citizen was found)"}: button says following: ${after.button}, camera following: ${after.camera} — ${String(after.camera) === after.button ? "they agree" : "THEY DISAGREE"}`);
    await context.close();
  }
  await browser.close();
})().catch((e) => { console.error(String(e).slice(0, 900)); process.exit(1); });
